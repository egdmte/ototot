━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

config.py:
  - 25 yeni parametre eklendi
  - Adaptif HSV profilleri: DARK, NORMAL, BRIGHT
  - KP: 0.40 → 0.60 | KD: 0.10 → 0.44 | BASE_SPEED: 65 → 62
  - Motor dengeleme: LEFT_TRIM_LOW/HIGH, RIGHT_TRIM_LOW/HIGH
  - Ölü bölge telafisi: DEAD_ZONE_PERCENT=20%, DEAD_ZONE_MIN_PWM=30

lane.py:
  - v_mean = np.mean(hsv[:, :, 2]) → Parlaklık hesabı
  - İf-elif-else ile profil seçimi
  - Debug çıktısında V değeri gösterilir

controller.py:
  - compute() → Hız-viraj koordinasyonu + dinamik kazanç
  - _apply_dead_zone_compensation() → Motor telafisi
  - _apply_speed_dependent_trim() → Trim seçimi
